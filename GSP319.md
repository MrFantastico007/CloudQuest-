
Copy the complete Bash script below and paste it into the Nano editor.

## Step 2: Paste the Automation Script

```bash
#!/usr/bin/env bash
set -euo pipefail

# ==============================================================================
# 1. USER-EDITABLE VARIABLES
# ==============================================================================
PROJECT_ID="$(gcloud config get-value project 2>/dev/null)"
ZONE="europe-west1-d"
REGION="${ZONE%-*}"

# Update these identifiers to match your Qwiklabs instructions.
CLUSTER_NAME="fancy-cluster-136"
MONOLITH_IDENTIFIER="fancy-monolith-525"
ORDERS_IDENTIFIER="fancy-orders-554"
PRODUCTS_IDENTIFIER="fancy-products-358"
FRONTEND_IDENTIFIER="fancy-frontend-187"
VERSION="1.0.0"

# ==============================================================================
# VALIDATION
# ==============================================================================
if [[ -z "$PROJECT_ID" || "$PROJECT_ID" == "(unset)" ]]; then
    echo "ERROR: No Google Cloud project is configured."
    exit 1
fi

echo "Project: $PROJECT_ID"
echo "Zone: $ZONE"
echo "Region: $REGION"

# ==============================================================================
# CONFIGURE GOOGLE CLOUD
# ==============================================================================
gcloud config set project "$PROJECT_ID"
gcloud config set compute/zone "$ZONE"
gcloud config set compute/region "$REGION"

# ==============================================================================
# ENABLE REQUIRED APIS WITH RETRIES
# ==============================================================================
echo "Enabling required APIs..."

API_ATTEMPT=1
MAX_ATTEMPTS=5

until gcloud services enable \
    container.googleapis.com \
    cloudbuild.googleapis.com \
    artifactregistry.googleapis.com
do
    if [[ "$API_ATTEMPT" -ge "$MAX_ATTEMPTS" ]]; then
        echo "ERROR: API enablement failed after $MAX_ATTEMPTS attempts."
        exit 1
    fi

    API_ATTEMPT=$((API_ATTEMPT + 1))
    echo "API enablement failed. Retrying in 30 seconds..."
    sleep 30
done

# ==============================================================================
# TASK 1: DOWNLOAD SOURCE AND BUILD MONOLITH
# ==============================================================================
echo "Task 1: Preparing the monolith application..."

cd "$HOME"

if [[ ! -d "$HOME/monolith-to-microservices/.git" ]]; then
    if [[ -d "$HOME/monolith-to-microservices" ]]; then
        echo "ERROR: Existing source directory lacks Git metadata."
        echo "Inspect or rename it before rerunning."
        exit 1
    fi

    git clone \
        https://github.com/googlecodelabs/monolith-to-microservices.git
fi

cd "$HOME/monolith-to-microservices"

# Install NVM if necessary.
if [[ ! -s "$HOME/.nvm/nvm.sh" ]]; then
    echo "Installing NVM..."
    curl -fsSL \
        https://raw.githubusercontent.com/nvm-sh/nvm/v0.39.7/install.sh \
        | bash
fi

export NVM_DIR="$HOME/.nvm"
source "$NVM_DIR/nvm.sh"

nvm install 22
nvm use 22

./setup.sh

# Build and push the monolith image.
cd "$HOME/monolith-to-microservices/monolith"

gcloud builds submit \
    --tag "gcr.io/${PROJECT_ID}/${MONOLITH_IDENTIFIER}:${VERSION}" \
    .

# ==============================================================================
# TASK 2: CREATE GKE CLUSTER AND DEPLOY MONOLITH
# ==============================================================================
echo "Task 2: Creating the GKE cluster..."

if ! gcloud container clusters describe "$CLUSTER_NAME" \
    --zone "$ZONE" >/dev/null 2>&1; then

    gcloud container clusters create "$CLUSTER_NAME" \
        --zone "$ZONE" \
        --num-nodes 3 \
        --machine-type e2-medium
else
    echo "Cluster already exists."
fi

gcloud container clusters get-credentials "$CLUSTER_NAME" \
    --zone "$ZONE"

if ! kubectl get deployment "$MONOLITH_IDENTIFIER" >/dev/null 2>&1; then
    kubectl create deployment "$MONOLITH_IDENTIFIER" \
        --image="gcr.io/${PROJECT_ID}/${MONOLITH_IDENTIFIER}:${VERSION}"
else
    kubectl set image \
        "deployment/${MONOLITH_IDENTIFIER}" \
        "${MONOLITH_IDENTIFIER}=gcr.io/${PROJECT_ID}/${MONOLITH_IDENTIFIER}:${VERSION}"
fi

if ! kubectl get service "$MONOLITH_IDENTIFIER" >/dev/null 2>&1; then
    kubectl expose deployment "$MONOLITH_IDENTIFIER" \
        --type=LoadBalancer \
        --port=80 \
        --target-port=8080
fi

# ==============================================================================
# TASKS 3 AND 4: BUILD AND DEPLOY ORDERS AND PRODUCTS
# ==============================================================================
echo "Tasks 3 and 4: Building microservices..."

cd "$HOME/monolith-to-microservices/microservices/src/orders"

gcloud builds submit \
    --tag "gcr.io/${PROJECT_ID}/${ORDERS_IDENTIFIER}:${VERSION}" \
    .

cd "$HOME/monolith-to-microservices/microservices/src/products"

gcloud builds submit \
    --tag "gcr.io/${PROJECT_ID}/${PRODUCTS_IDENTIFIER}:${VERSION}" \
    .

# Deploy Orders.
if ! kubectl get deployment "$ORDERS_IDENTIFIER" >/dev/null 2>&1; then
    kubectl create deployment "$ORDERS_IDENTIFIER" \
        --image="gcr.io/${PROJECT_ID}/${ORDERS_IDENTIFIER}:${VERSION}"
else
    kubectl set image \
        "deployment/${ORDERS_IDENTIFIER}" \
        "${ORDERS_IDENTIFIER}=gcr.io/${PROJECT_ID}/${ORDERS_IDENTIFIER}:${VERSION}"
fi

if ! kubectl get service "$ORDERS_IDENTIFIER" >/dev/null 2>&1; then
    kubectl expose deployment "$ORDERS_IDENTIFIER" \
        --type=LoadBalancer \
        --port=80 \
        --target-port=8081
fi

# Deploy Products.
if ! kubectl get deployment "$PRODUCTS_IDENTIFIER" >/dev/null 2>&1; then
    kubectl create deployment "$PRODUCTS_IDENTIFIER" \
        --image="gcr.io/${PROJECT_ID}/${PRODUCTS_IDENTIFIER}:${VERSION}"
else
    kubectl set image \
        "deployment/${PRODUCTS_IDENTIFIER}" \
        "${PRODUCTS_IDENTIFIER}=gcr.io/${PROJECT_ID}/${PRODUCTS_IDENTIFIER}:${VERSION}"
fi

if ! kubectl get service "$PRODUCTS_IDENTIFIER" >/dev/null 2>&1; then
    kubectl expose deployment "$PRODUCTS_IDENTIFIER" \
        --type=LoadBalancer \
        --port=80 \
        --target-port=8082
fi

# ==============================================================================
# WAIT FOR EXTERNAL IP ADDRESSES
# ==============================================================================
get_ip() {
    local service="$1"
    local ip=""
    local attempt

    for attempt in $(seq 1 60); do
        ip="$(kubectl get service "$service" \
            -o jsonpath='{.status.loadBalancer.ingress[0].ip}' \
            2>/dev/null || true)"

        if [[ -n "$ip" && "$ip" != "<pending>" ]]; then
            echo "$ip"
            return 0
        fi

        echo "Waiting for $service external IP ($attempt/60)..." >&2
        sleep 10
    done

    echo "ERROR: Timed out waiting for $service external IP." >&2
    return 1
}

echo "Waiting for Orders and Products IP addresses..."

ORDERS_IP="$(get_ip "$ORDERS_IDENTIFIER")"
PRODUCTS_IP="$(get_ip "$PRODUCTS_IDENTIFIER")"

echo "Orders IP: $ORDERS_IP"
echo "Products IP: $PRODUCTS_IP"

# ==============================================================================
# TASK 5: CONFIGURE FRONTEND
# ==============================================================================
echo "Task 5: Configuring frontend..."

cd "$HOME/monolith-to-microservices/react-app"

cat > .env <<EOF
REACT_APP_ORDERS_URL=http://${ORDERS_IP}/api/orders
REACT_APP_PRODUCTS_URL=http://${PRODUCTS_IP}/api/products
EOF

npm run build

# ==============================================================================
# TASK 6: BUILD FRONTEND CONTAINER
# ==============================================================================
echo "Task 6: Building frontend container..."

cd "$HOME/monolith-to-microservices/microservices/src/frontend"

gcloud builds submit \
    --tag "gcr.io/${PROJECT_ID}/${FRONTEND_IDENTIFIER}:${VERSION}" \
    .

# ==============================================================================
# TASK 7: DEPLOY FRONTEND
# ==============================================================================
echo "Task 7: Deploying frontend..."

if ! kubectl get deployment "$FRONTEND_IDENTIFIER" >/dev/null 2>&1; then
    kubectl create deployment "$FRONTEND_IDENTIFIER" \
        --image="gcr.io/${PROJECT_ID}/${FRONTEND_IDENTIFIER}:${VERSION}"
else
    kubectl set image \
        "deployment/${FRONTEND_IDENTIFIER}" \
        "${FRONTEND_IDENTIFIER}=gcr.io/${PROJECT_ID}/${FRONTEND_IDENTIFIER}:${VERSION}"
fi

if ! kubectl get service "$FRONTEND_IDENTIFIER" >/dev/null 2>&1; then
    kubectl expose deployment "$FRONTEND_IDENTIFIER" \
        --type=LoadBalancer \
        --port=80 \
        --target-port=8080
fi

echo "Waiting for frontend external IP..."

FRONTEND_IP="$(get_ip "$FRONTEND_IDENTIFIER")"

echo
echo "=============================================="
echo "Deployment script finished."
echo "Frontend URL: http://${FRONTEND_IP}"
echo "=============================================="

kubectl get deployments
kubectl get services

echo "Return to the lab and click Check my progress."
```

## Troubleshooting

**If the script stops:** Read the first error message in Cloud Shell. Because `set -euo pipefail` is enabled, the script exits when a command fails instead of silently continuing.

**If an IP remains pending:**

```bash
kubectl get services
kubectl describe service ORDERS_IDENTIFIER
```

Replace `ORDERS_IDENTIFIER` with the actual service name.

**If the checker does not pass:** Confirm that the resource names, image names, zone, ports, and required configurations match the exact challenge-lab instructions. A working website alone does not guarantee every task is satisfied.

**Important:** This script uses `gcr.io` image paths. If your lab specifically requires Artifact Registry repository paths or different deployment settings, follow those requirements instead.
