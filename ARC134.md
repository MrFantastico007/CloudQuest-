#!/usr/bin/env bash
# ==============================================================================
# Google Cloud Lab Automation Script
# Lab ARC134: Configure Service Accounts and IAM Roles for Google Cloud
# ==============================================================================

set -euo pipefail

# ------------------------------------------------------------------------------
# 1. USER-EDITABLE VARIABLES (Update these values with your lab details)
# ------------------------------------------------------------------------------
PROJECT_ID="qwiklabs-gcp-02-6c894e3cd290"
REGION="us-east4"
ZONE="us-east4-c"

# Service Accounts & Compute Instances Configuration
DEVOPS_SA_NAME="devops"
DEVOPS_VM_NAME="vm-2"

BQ_SA_NAME="bigquery-qwiklab"
BQ_VM_NAME="bigquery-instance"

# Custom Role Configuration
CUSTOM_ROLE_ID="custom_cloudsql_role"
YAML_FILE="role-definition.yaml"

# ------------------------------------------------------------------------------
# 2. VALIDATION & ENVIRONMENT SETUP
# ------------------------------------------------------------------------------
echo "==> Validating configuration..."
if [[ -z "$PROJECT_ID" || -z "$REGION" || -z "$ZONE" ]]; then
    echo "ERROR: PROJECT_ID, REGION, and ZONE must be set." >&2
    exit 1
fi

echo "==> Configuring gcloud defaults..."
gcloud config set project "$PROJECT_ID"
gcloud config set compute/region "$REGION"
gcloud config set compute/zone "$ZONE"

DEVOPS_SA_EMAIL="${DEVOPS_SA_NAME}@${PROJECT_ID}.iam.gserviceaccount.com"
BQ_SA_EMAIL="${BQ_SA_NAME}@${PROJECT_ID}.iam.gserviceaccount.com"

# ------------------------------------------------------------------------------
# TASK 2: Create a service account named 'devops'
# ------------------------------------------------------------------------------
echo "==> Task 2: Creating service account '${DEVOPS_SA_NAME}'..."
if ! gcloud iam service-accounts describe "$DEVOPS_SA_EMAIL" &>/dev/null; then
    gcloud iam service-accounts create "$DEVOPS_SA_NAME" \
        --display-name="DevOps Service Account"
else
    echo "Service account '${DEVOPS_SA_NAME}' already exists. Skipping creation."
fi

# ------------------------------------------------------------------------------
# TASK 3: Grant IAM permissions to the 'devops' service account
# ------------------------------------------------------------------------------
echo "==> Task 3: Granting IAM permissions to '${DEVOPS_SA_EMAIL}'..."
gcloud projects add-iam-policy-binding "$PROJECT_ID" \
    --member="serviceAccount:${DEVOPS_SA_EMAIL}" \
    --role="roles/iam.serviceAccountUser"

gcloud projects add-iam-policy-binding "$PROJECT_ID" \
    --member="serviceAccount:${DEVOPS_SA_EMAIL}" \
    --role="roles/compute.instanceAdmin.v1"

# ------------------------------------------------------------------------------
# TASK 4: Create a compute instance with the 'devops' service account attached
# ------------------------------------------------------------------------------
echo "==> Task 4: Creating Compute Instance '${DEVOPS_VM_NAME}'..."
if ! gcloud compute instances describe "$DEVOPS_VM_NAME" --zone="$ZONE" &>/dev/null; then
    gcloud compute instances create "$DEVOPS_VM_NAME" \
        --zone="$ZONE" \
        --machine-type="e2-medium" \
        --service-account="$DEVOPS_SA_EMAIL" \
        --scopes="cloud-platform"
else
    echo "Compute instance '${DEVOPS_VM_NAME}' already exists. Skipping creation."
fi

# ------------------------------------------------------------------------------
# TASK 5: Create a custom role using a YAML file
# ------------------------------------------------------------------------------
echo "==> Task 5: Creating custom role using '${YAML_FILE}'..."
cat <<EOF > "$YAML_FILE"
title: "Custom CloudSQL Role"
description: "Custom role for CloudSQL connectivity and inspection"
stage: "GA"
includedPermissions:
  - cloudsql.instances.connect
  - cloudsql.instances.get
EOF

if ! gcloud iam roles describe "$CUSTOM_ROLE_ID" --project="$PROJECT_ID" &>/dev/null; then
    gcloud iam roles create "$CUSTOM_ROLE_ID" \
        --project="$PROJECT_ID" \
        --file="$YAML_FILE"
else
    echo "Custom role '${CUSTOM_ROLE_ID}' already exists. Updating existing role..."
    gcloud iam roles update "$CUSTOM_ROLE_ID" \
        --project="$PROJECT_ID" \
        --file="$YAML_FILE"
fi

# ------------------------------------------------------------------------------
# TASK 6: Use client libraries to access BigQuery from a service account
# ------------------------------------------------------------------------------
echo "==> Task 6: Setting up BigQuery service account, VM instance, and query script..."

# 1. Create service account
if ! gcloud iam service-accounts describe "$BQ_SA_EMAIL" &>/dev/null; then
    gcloud iam service-accounts create "$BQ_SA_NAME" \
        --display-name="BigQuery Service Account"
else
    echo "Service account '${BQ_SA_NAME}' already exists. Skipping creation."
fi

# 2. Assign required roles
gcloud projects add-iam-policy-binding "$PROJECT_ID" \
    --member="serviceAccount:${BQ_SA_EMAIL}" \
    --role="roles/bigquery.dataViewer"

gcloud projects add-iam-policy-binding "$PROJECT_ID" \
    --member="serviceAccount:${BQ_SA_EMAIL}" \
    --role="roles/bigquery.user"

# 3. Create VM Instance attached to BigQuery service account
if ! gcloud compute instances describe "$BQ_VM_NAME" --zone="$ZONE" &>/dev/null; then
    gcloud compute instances create "$BQ_VM_NAME" \
        --zone="$ZONE" \
        --machine-type="e2-medium" \
        --service-account="$BQ_SA_EMAIL" \
        --scopes="cloud-platform"
else
    echo "Compute instance '${BQ_VM_NAME}' already exists. Skipping creation."
fi

# 4. Generate Python client script locally
cat <<EOF > query.py
from google.auth import compute_engine
from google.cloud import bigquery

credentials = compute_engine.Credentials(
    service_account_email='${BQ_SA_EMAIL}'
)

query = '''
SELECT name, SUM(number) as total_people
FROM \`bigquery-public-data.usa_names.usa_1910_2013\`
WHERE state = 'TX'
GROUP BY name, state
ORDER BY total_people DESC
LIMIT 20
'''

client = bigquery.Client(
    project='${PROJECT_ID}',
    credentials=credentials
)

print(client.query(query).to_dataframe())
EOF

echo "=============================================================================="
echo "SUCCESS: All lab automated tasks have completed!"
echo "=============================================================================="

SERVICE_ACCOUNT_EMAIL = f"bigquery-qwiklab@{PROJECT_ID}.iam.gserviceaccount.com"

credentials = compute_engine.Credentials(
    service_account_email=SERVICE_ACCOUNT_EMAIL
)

query = '''
SELECT name, SUM(number) as total_people
FROM `bigquery-public-data.usa_names.usa_1910_2013`
WHERE state = 'TX'
GROUP BY name, state
ORDER BY total_people DESC
LIMIT 20
'''

client = bigquery.Client(
    project=PROJECT_ID,
    credentials=credentials
)

EOF
