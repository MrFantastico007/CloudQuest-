## Step 2: Paste the complete script

Paste **only the Bash script below** into `nano`. Do not include the Markdown code fences or the README instructions.

```bash
#!/usr/bin/env bash
set -Eeuo pipefail

# ==============================================================================
# 1. USER-EDITABLE VARIABLES
# ==============================================================================

export PROJECT_ID="qwiklabs-gcp-01-7dd1de9b75ea"
export REGION="europe-west4"
export TOPIC_NAME="memories-topic-955"

export BUCKET_NAME="memories-bucket-${PROJECT_ID}"
export FUNCTION_NAME="memories-thumbnail-creator"
export ENTRY_POINT="memories-thumbnail-creator"
export RUNTIME="nodejs22"

# ==============================================================================
# 2. HELPER FUNCTIONS
# ==============================================================================

log() {
  echo
  echo "==> $*"
}

die() {
  echo "ERROR: $*" >&2
  exit 1
}

command -v gcloud >/dev/null 2>&1 || die "gcloud is not installed."
command -v curl >/dev/null 2>&1 || die "curl is not installed."

# ==============================================================================
# 3. PROJECT SETUP AND API ENABLEMENT
# ==============================================================================

log "Setting active project: ${PROJECT_ID}"
gcloud config set project "${PROJECT_ID}"

log "Enabling required Google Cloud APIs"
gcloud services enable \
  storage.googleapis.com \
  pubsub.googleapis.com \
  cloudfunctions.googleapis.com \
  run.googleapis.com \
  eventarc.googleapis.com \
  cloudbuild.googleapis.com \
  artifactregistry.googleapis.com \
  --project="${PROJECT_ID}"

log "Retrieving project number"
PROJECT_NUMBER="$(
  gcloud projects describe "${PROJECT_ID}" \
    --format="value(projectNumber)"
)"

[[ -n "${PROJECT_NUMBER}" ]] || die "Could not retrieve project number."

COMPUTE_SA="${PROJECT_NUMBER}-compute@developer.gserviceaccount.com"
EVENTARC_SA="service-${PROJECT_NUMBER}@gcp-sa-eventarc.iam.gserviceaccount.com"

# Initialize Google-managed service identities.
log "Initializing Google-managed service identities"

gcloud beta services identity create \
  --service=eventarc.googleapis.com \
  --project="${PROJECT_ID}" || true

gcloud beta services identity create \
  --service=cloudfunctions.googleapis.com \
  --project="${PROJECT_ID}" || true

gcloud beta services identity create \
  --service=pubsub.googleapis.com \
  --project="${PROJECT_ID}" || true

# ==============================================================================
# 4. IAM PERMISSIONS
# ==============================================================================

log "Waiting for the Cloud Storage service agent"

STORAGE_SA=""
for attempt in $(seq 1 12); do
  STORAGE_SA="$(
    gcloud storage service-agent \
      --project="${PROJECT_ID}" \
      --format="value(email)" 2>/dev/null || true
  )"

  if [[ -n "${STORAGE_SA}" ]]; then
    break
  fi

  echo "Waiting for Storage service agent (${attempt}/12)..."
  sleep 10
done

if [[ -z "${STORAGE_SA}" ]]; then
  echo "WARNING: Storage service agent is not available yet."
fi

grant_role() {
  local member="$1"
  local role="$2"
  local attempt

  for attempt in $(seq 1 6); do
    if gcloud projects add-iam-policy-binding "${PROJECT_ID}" \
      --member="serviceAccount:${member}" \
      --role="${role}" \
      --condition=None \
      --quiet >/dev/null 2>&1; then
      echo "Granted ${role} to ${member}"
      return 0
    fi

    echo "IAM grant pending/failed: ${role} -> ${member} (${attempt}/6)"
    sleep 10
  done

  echo "WARNING: Could not grant ${role} to ${member}."
  return 0
}

if [[ -n "${STORAGE_SA}" ]]; then
  grant_role "${STORAGE_SA}" "roles/pubsub.publisher"
fi

grant_role "${EVENTARC_SA}" "roles/eventarc.serviceAgent"
grant_role "${COMPUTE_SA}" "roles/eventarc.eventReceiver"
grant_role "${COMPUTE_SA}" "roles/pubsub.publisher"

log "Allowing time for IAM propagation"
sleep 30

# ==============================================================================
# 5. TASK 1: CREATE CLOUD STORAGE BUCKET
# ==============================================================================

log "Task 1: Checking Cloud Storage bucket"

if gcloud storage buckets describe "gs://${BUCKET_NAME}" \
  --project="${PROJECT_ID}" >/dev/null 2>&1; then
  echo "Bucket already exists: gs://${BUCKET_NAME}"
else
  gcloud storage buckets create "gs://${BUCKET_NAME}" \
    --project="${PROJECT_ID}" \
    --location="${REGION}" ||
    die "Could not create bucket. Check your lab permissions and bucket name."
fi

# ==============================================================================
# 6. TASK 2: CREATE PUB/SUB TOPIC
# ==============================================================================

log "Task 2: Checking Pub/Sub topic"

if gcloud pubsub topics describe "${TOPIC_NAME}" \
  --project="${PROJECT_ID}" >/dev/null 2>&1; then
  echo "Topic already exists: ${TOPIC_NAME}"
else
  gcloud pubsub topics create "${TOPIC_NAME}" \
    --project="${PROJECT_ID}" ||
    die "Could not create Pub/Sub topic."
fi

# ==============================================================================
# 7. TASK 3: PREPARE CLOUD FUNCTION SOURCE
# ==============================================================================

log "Task 3: Preparing Cloud Function source"

SOURCE_DIR="${HOME}/${FUNCTION_NAME}_src"
mkdir -p "${SOURCE_DIR}"

cat > "${SOURCE_DIR}/package.json" <<'PACKAGE_JSON'
{
  "name": "thumbnails",
  "version": "1.0.0",
  "description": "Create thumbnails of uploaded images",
  "main": "index.js",
  "scripts": {
    "start": "node index.js"
  },
  "dependencies": {
    "@google-cloud/functions-framework": "^3.0.0",
    "@google-cloud/pubsub": "^2.0.0",
    "@google-cloud/storage": "^6.11.0",
    "sharp": "^0.32.1"
  },
  "engines": {
    "node": ">=18"
  }
}
PACKAGE_JSON

cat > "${SOURCE_DIR}/index.js" <<'FUNCTION_JS'
const functions = require('@google-cloud/functions-framework');
const { Storage } = require('@google-cloud/storage');
const { PubSub } = require('@google-cloud/pubsub');
const sharp = require('sharp');

const storage = new Storage();
const pubsub = new PubSub();
const topicName = process.env.TOPIC_NAME || 'memories-topic-955';

functions.cloudEvent('memories-thumbnail-creator', async cloudEvent => {
  const event = cloudEvent.data || {};
  const fileName = event.name;
  const bucketName = event.bucket;

  console.log(`Event: ${JSON.stringify(event)}`);

  if (!fileName || !bucketName) {
    console.error('Missing file name or bucket in event.');
    return;
  }

  if (fileName.includes('64x64_thumbnail')) {
    console.log(`Thumbnail already exists: ${fileName}`);
    return;
  }

  const extension = fileName.split('.').pop().toLowerCase();

  if (!['png', 'jpg', 'jpeg'].includes(extension)) {
    console.log(`Unsupported image: ${fileName}`);
    return;
  }

  const bucket = storage.bucket(bucketName);
  const original = bucket.file(fileName);
  const baseName = fileName.slice(0, -(extension.length + 1));
  const thumbnailName = `${baseName}_64x64_thumbnail.${extension}`;

  try {
    const [buffer] = await original.download();

    const thumbnailBuffer = await sharp(buffer)
      .resize(64, 64, {
        fit: 'inside',
        withoutEnlargement: true
      })
      .toFormat(extension)
      .toBuffer();

    await bucket.file(thumbnailName).save(thumbnailBuffer, {
      resumable: false,
      metadata: {
        contentType: `image/${extension === 'jpg' ? 'jpeg' : extension}`
      }
    });

    console.log(`Created thumbnail: ${thumbnailName}`);

    await pubsub.topic(topicName).publishMessage({
      data: Buffer.from(thumbnailName)
    });

    console.log(`Published message to ${topicName}`);
  } catch (error) {
    console.error('Thumbnail processing failed:', error);
    throw error;
  }
});
FUNCTION_JS

log "Deploying Cloud Functions Gen 2"

gcloud functions deploy "${FUNCTION_NAME}" \
  --gen2 \
  --project="${PROJECT_ID}" \
  --region="${REGION}" \
  --runtime="${RUNTIME}" \
  --source="${SOURCE_DIR}" \
  --entry-point="${ENTRY_POINT}" \
  --trigger-event-filters="type=google.cloud.storage.object.v1.finalized" \
  --trigger-event-filters="bucket=${BUCKET_NAME}" \
  --trigger-location="${REGION}" \
  --set-env-vars="TOPIC_NAME=${TOPIC_NAME}" \
  --quiet ||
  die "Function deployment failed. Review the error above, especially IAM permissions."

# ==============================================================================
# 8. TASK 4: UPLOAD SAMPLE IMAGE
# ==============================================================================

log "Task 4: Downloading sample image"

cd "${HOME}"

curl -fL --retry 3 \
  "https://storage.googleapis.com/cloud-training/arc101/travel.jpg" \
  -o travel.jpg ||
  die "Could not download travel.jpg."

log "Uploading sample image to Cloud Storage"

gcloud storage cp travel.jpg "gs://${BUCKET_NAME}/" ||
  die "Image upload failed."

# ==============================================================================
# 9. VERIFY RESULTS
# ==============================================================================

log "Uploaded objects"
gcloud storage ls "gs://${BUCKET_NAME}/"

log "Deployed function"
gcloud functions describe "${FUNCTION_NAME}" \
  --gen2 \
  --project="${PROJECT_ID}" \
  --region="${REGION}" \
  --format="yaml(name,state,serviceConfig.uri,eventTrigger.eventType)" ||
  echo "WARNING: Could not retrieve function details."

echo
echo "=============================================================="
echo "Script finished."
echo "Bucket:   gs://${BUCKET_NAME}"
echo "Topic:    ${TOPIC_NAME}"
echo "Function: ${FUNCTION_NAME}"
echo
echo "Wait for the storage event to process the image."
echo "Expected thumbnail: travel_64x64_thumbnail.jpg"
echo "=============================================================="
```

## Important notes

- Confirm that the project ID, region, and Pub/Sub topic at the top of the script match your active lab session.
- The script attempts IAM role grants, but it cannot override lab or organization policy restrictions.
- Allow time for the Cloud Storage event trigger to process the uploaded image. A successful upload does not, by itself, confirm that thumbnail generation succeeded.
